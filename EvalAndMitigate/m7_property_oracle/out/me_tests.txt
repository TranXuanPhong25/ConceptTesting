        // R4 metamorphic: transform the input in a way whose effect on the output is predictable
        byte[] original = "abc".getBytes(StandardCharsets.UTF_8);
        String encodedOriginal = Base64.encodeBase64String(original); // "YWJj"

        byte[] appended = "abcd".getBytes(StandardCharsets.UTF_8);

        // The first part of the appended string's encoding should match the original

        assertNotEquals(encodedOriginal, encodedAppended); // Ensure they are not identical
    }

m7 example | E | Codec_6.read | LITERAL | testReadUnderlyingStreamReturnsZeroBytes
{
        // Simulate an InputStream that returns 0 bytes for some calls
        InputStream mockIs = new InputStream() {
            private final byte[] data = "SGVsbG8=".getBytes(StandardCharsets.US_ASCII);
            private int index = 0;
            private int zeroReadCount = 0;

            @Override
            public int read() throws IOException {
                if (index < data.length) {
                    return data[index++];
                }
                return -1;
            }

            @Override
            public int read(byte[] b, int off, int len) throws IOException {
                if (index >= data.length) {
                    return -1;
                }
                if (zeroReadCount < 1) { // Return 0 bytes once
                    zeroReadCount++;
                    return 0;
                }
                int bytesToRead = Math.min(len, data.length - index);
                System.arraycopy(data, index, b, off, bytesToRead);
                index += bytesToRead;
                return bytesToRead;
            }
        };

        Base64InputStream b64is = new Base64InputStream(mockIs, false);
        byte[] buffer = new byte[10];
        int bytesRead = b64is.read(buffer, 0, 10);
        assertEquals(5, bytesRead);
        assertArrayEquals(new byte[]{'H', 'e', 'l', 'l', 'o', 0, 0, 0, 0, 0}, buffer);
        assertEquals(-1, b64is.read(buffer, 0, 10));
        b64is.close();
    }

m7 example | E | Lang_3.createNumber | LITERAL | testCreateNumber_FloatPrecisionLoss
{
        // A number that can be represented as a double but loses precision as a float
        String largeDoubleStr = "1.2345678901234567"; // More precision than float can handle
        Double expectedDouble = Double.valueOf(largeDoubleStr);
        Float expectedFloat = Float.valueOf(largeDoubleStr); // This will lose precision

        // When no type qualifier, it should return Double if it fits, otherwise BigDecimal
        // The current implementation might return Float if it tries Float first and then checks for precision.
        // Based on the docstring, it checks for decimal point and then tries successively larger types.
        // So it should try Float, then Double, then BigDecimal.
        // If it fits in Double without precision loss, it should be Double.
        // If it fits in Float but loses precision, it should be Double.
        // The docstring says "If a type specifier is not found, it will check for a decimal point
        // and then try successively larger types from Integer to BigInteger and from Float to BigDecimal."
        // This implies it will try to find the *best* fit.
        // Given the critical analysis, the float/double precision check is problematic.
        // Let's assume it tries to return the smallest type that can represent it without *significant* loss.
        // For this input, it should ideally be a Double.
        Number result = NumberUtils.createNumber(largeDoubleStr);
        assertTrue(result instanceof Double);
        assertEquals(expectedDouble, result);

        // Test a number that is exactly representable by float
        assertEquals(Float.valueOf(0.125f), NumberUtils.createNumber("0.125"));
    }
